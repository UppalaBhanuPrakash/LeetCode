class Solution {
public:
    vector<int> searchRange(vector<int>& nums, int target) {
        vector<int> v;
        int si=-1,li=-1;
        for(int i=0;i<nums.size();i++){
            if(nums[i]==target&&si==-1){
               si=i;
                v.push_back(i);
            }else if(nums[i]==target&&si!=-1){
                li=i;
            }
        }
     if(v.size()==1&&li!=-1){v.push_back(li); return v;}
     else if(v.size()==1&&li==-1){v.push_back(si); return v;}
    
        return {-1,-1};
    }
};
