class Solution {
public:
    int numSteps(string s) {
        int carry=0;
        int c=0;
        for(int i=s.length()-1;i>=0;i--){
            if(i==0&&s[i]=='1')break;
            if(s[i]=='1'&&carry==0){
                c+=2;
                carry=1;
            }
            else if(s[i]=='1'&&carry==1){
                c+=1;
                carry=1;
            }
            else if(s[i]=='0'&&carry==0){
                c+=1;
            }
            else{
                c+=2;
                carry=1;
            }
        }
        if(s[0]=='1'&&carry==1)c+=1;
        return c;
    }
};
