class Solution {
public:
    vector<int> toggleLightBulbs(vector<int>& bulbs) {
        map<int,int> mp;
        vector<int> v;
        for(int i=0;i<bulbs.size();i++){
            mp[bulbs[i]]++;
        }
        for(auto it:mp){
            if(it.second%2!=0){
                v.push_back(it.first);
            }

        }
        sort(v.begin(),v.end());
        return v;
    }
};
