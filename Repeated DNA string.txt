class Solution {
public:
    vector<string> findRepeatedDnaSequences(string s) {
        map<string,int> mp;
        vector<string> v;
        if(s.length()<10)return v;
        
        for(int i=0;i<s.length()-9;i++){
            string s1=s.substr(i,10);
            if(mp.find(s1)!=mp.end()){
                mp[s1]++;
            }
            else{
                mp[s1]=1;
            }
        }
        for(auto it:mp){
            if(it.second>1){
                v.push_back(it.first);
            }
        }
        return v;
    }
};
