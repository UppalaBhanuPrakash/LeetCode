class Solution {
public:
    int countPrimeSetBits(int left, int right) {
        int ans=0;
        for(int i=left;i<=right;i++){
            int c=0;
            int n=i;
            while(n>0){
                if(n%2!=0)
                c++;
                n=n/2;
            }
            // int p=1;
            // for(int i=2;i<=sqrt(c);i++){
            //     if(c%i==0)p++;
            // }
            // if(p==1&&c!=1)ans++;
          if(c==2||c==3||c==5||c==7||c==11||c==13||c==17||c==19)ans++;

        }
        return ans;
    }
};
