class Solution {
public:
    bool isPalindrome(string s) {
         int n=s.size();
        string str="";
        for(int i=0;i<n;i++){
            if((s[i]>='a' && s[i]<='z')||(s[i]>='A' && s[i]<='Z')||(s[i]>='0' && s[i]<='9') ){
                str+=s[i];
            }
        }
        transform(str.begin(),str.end(),str.begin(),::tolower);
        string str2=str;
        reverse(str.begin(),str.end());
        return str==str2;
    }
};
