class Solution {
public:
    int reverseBits(int n) {
        string s;
        int res=0;
        while(n>0){
            int rem=n%2;
            n=n/2;
            s+=to_string(rem);
        }
        while(s.length()<32){
            s+='0';
        }
        int j=0;
        for(int i=s.length()-1;i>=0;i--){
            if(s[i]=='1')
            res+=pow(2,j);
            j++;
        } 
        return res;
    }
};
