class Solution {
public:
    vector<int> sortByBits(vector<int>& arr) {
        map<int,vector<int>> mp;
        vector<int> v;
        for(int i=0;i<arr.size();i++){
            int curr=arr[i],c=0;
            while(curr>0){
               c += curr % 2;   
               curr /= 2;
            }
            mp[c].push_back(arr[i]);
        }
           for(auto p : mp){
            sort(p.second.begin(), p.second.end()); 
            for(int num : p.second){
                v.push_back(num);
            }
        }

        return v;
    }
};
