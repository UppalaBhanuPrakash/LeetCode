class Solution {
public:
    vector<int> constructTransformedArray(vector<int>& nums) {
         int n = nums.size();
    vector<int> result(n);

    for (int i = 0; i < n; i++) {
        if (nums[i] > 0) {
            int ans = nums[i];
            result[i] = nums[(i + ans) % n];
        }
        else if (nums[i] < 0) {
            int ans = abs(nums[i]) % n;
            result[i] = nums[(i - ans + n) % n];
        }
        else {
            result[i] = nums[i];
        }
    }
    return result;
    }
};
